% --- Alt text for math -------------------------------------------------------
% math/alt/use = true (in \DocumentMetadata) gives every formula alt text
% automatically: its LaTeX source. \mathalt{words}{formula} replaces that
% with words you write, for this one formula. Plain words only, no macros.
% It sets a variable inside the LaTeX tagging code; if a later LaTeX renames
% that variable, formulas simply fall back to the automatic alt text.
\ExplSyntaxOn
\tl_if_exist:NF \l__math_content_template_tl { \tl_new:N \l__math_content_template_tl }
\NewDocumentCommand{\mathalt}{m +m}
  {
    \group_begin:
    \tl_set:Nn \l__math_content_template_tl { \exp_not:n {#1} }
    #2
    \group_end:
  }
\ExplSyntaxOff

% --- Pseudocode ------------------------------------------------------------
% \begin{pseudocode}{title}{label} ... \end{pseudocode}
% A numbered, titled, single-spaced block that does not float and is never
% split across pages (split a very long algorithm in two). Each line
% becomes a list item in the tag tree, so a screen reader can step
% through it.
\newcounter{algorithm}[chapter]
\renewcommand{\thealgorithm}{\thechapter.\arabic{algorithm}}
\makeatletter
\newenvironment{pseudocode}[2]{%
  \par\addvspace{\baselineskip}%
  \begingroup
  \setstretch{1}%
  % penalties that forbid a page break anywhere inside the block
  \interlinepenalty=\@M \@itempenalty=\@M \@beginparpenalty=\@M \@endparpenalty=\@M
  \hrule height 0.9pt \nobreak\vskip 5pt
  \noindent\refstepcounter{algorithm}\label{#2}%
  \textbf{Algorithm \thealgorithm.} #1\par
  \nobreak\vskip 5pt \hrule height 0.4pt \nobreak\vskip 4pt
  \begin{algorithmic}[1]}%
  {\end{algorithmic}\par
   \nobreak\vskip 4pt \hrule height 0.9pt
   \endgroup\par\addvspace{\baselineskip}}
\makeatother
\crefname{algorithm}{algorithm}{algorithms}
\Crefname{algorithm}{Algorithm}{Algorithms}
\algrenewcommand{\algorithmiccomment}[1]{\hfill\textit{// #1}}
